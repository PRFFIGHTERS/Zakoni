Попробуйте изменить формулировку.`; return; } empty.classList.add('hidden'); list.innerHTML = results.map((item, index) => { const snippetText = item.article.text.substring(0, 200).replace(/\n/g, ' ') + '...'; const highlightedSnippet = highlightText(snippetText, query); return `
${escapeHtml(item.docTitle)}
Статья ${escapeHtml(item.article.number)}. ${escapeHtml(item.article.title)}
${highlightedSnippet}
Совпадений: ${item.score} ${escapeHtml(item.article.status.split('.')[0])}
`; }).join(''); } function openSearchScreen(query) { AppState.searchQuery = query; const resultsInput = document.getElementById('searchResultsInput'); if (resultsInput) resultsInput.value = query; const searchData = performSearch(query); } // Статистика let totalArticles = 0; LAW_DOCUMENTS.forEach(doc => totalArticles += doc.articles.length); const statDocs = document.getElementById('statDocs'); const statArticles = document.getElementById('statArticles'); const statLastCheck = document.getElementById('statLastCheck'); if (statDocs) statDocs.textContent = LAW_DOCUMENTS.length; if (statArticles) statArticles.textContent = totalArticles; if (statLastCheck) { statLastCheck.textContent = AppState.lastCheck ? new Date(AppState.lastCheck).toLocaleDateString('ru-RU', { day: 'numeric', month: 'short', year: 'numeric' }) : 'Не проверялось'; } } function applyTheme() { document.documentElement.setAttribute('data-theme', AppState.theme); localStorage.setItem('app_theme', AppState.theme); } function applyFontSize() { document.documentElement.style.fontSize = (16 * AppState.fontSize / 100) + 'px'; localStorage.setItem('app_fontSize', AppState.fontSize); } // ===== LOCAL STORAGE ===== function saveFavorites() { localStorage.setItem('app_favorites', JSON.stringify(AppState.favorites)); } function loadState() { try { const savedTheme = localStorage.getItem('app_theme'); if (savedFavorites) AppState.favorites = JSON.parse(savedFavorites); const savedHistory = localStorage.getItem('app_history'); if (savedHistory) AppState.history = JSON.parse(savedHistory); } catch (e) { console.warn('Ошибка загрузки состояния:', e); } } // ===== ИНИЦИАЛИЗАЦИЯ И ОБРАБОТЧИКИ СОБЫТИЙ ===== document.addEventListener('DOMContentLoaded', () => { loadState(); applyTheme(); applyFontSize(); renderLawCards(); updateSettingsUI(); // === Навигация по нижней панели === document.querySelectorAll('.nav-item').forEach(item => { item.addEventListener('click', () => { switchScreen(item.dataset.screen); }); }); // === Главный поиск === const mainSearchInput = document.getElementById('mainSearchInput'); const mainClearBtn = document.getElementById('mainClearBtn'); if (mainSearchInput) { mainSearchInput.addEventListener('input', (e) => { if (mainClearBtn) mainClearBtn.classList.toggle('visible', e.target.value.length > 0); mainClearBtn.classList.remove('visible'); mainSearchInput.focus(); }); } // === Поиск на экране результатов === const searchResultsInput = document.getElementById('searchResultsInput'); const searchResultsClearBtn = document.getElementById('searchResultsClearBtn');
